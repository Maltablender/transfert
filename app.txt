function Format-Size([double]$b) {
    if ($b -ge 1e12) { return ('{0:N2} To' -f ($b / 1e12)) }
    if ($b -ge 1e9)  { return ('{0:N1} Go' -f ($b / 1e9)) }
    if ($b -ge 1e6)  { return ('{0:N1} Mo' -f ($b / 1e6)) }
    if ($b -ge 1e3)  { return ('{0:N0} Ko' -f ($b / 1e3)) }
    return ('{0:N0} o' -f $b)
}
function Format-Dur([double]$s) {
    if ($s -lt 0 -or [double]::IsInfinity($s) -or [double]::IsNaN($s)) { return '–' }
    if ($s -lt 60) { return ('{0:N0} s' -f [math]::Max(1, $s)) }
    if ($s -lt 3600) { return ('{0:N0} min' -f [math]::Ceiling($s / 60)) }
    $h = [math]::Floor($s / 3600); $m = [math]::Floor(($s % 3600) / 60)
    return ('{0} h {1:00}' -f $h, $m)
}
function ConvertTo-Bytes([string]$n, [string]$u) {
    $v = [double]::Parse($n, [Globalization.CultureInfo]::InvariantCulture)
    switch -regex ($u) {
        '^[kK]iB$' { return $v * 1024 }
        '^MiB$'    { return $v * 1048576 }
        '^GiB$'    { return $v * 1073741824 }
        '^TiB$'    { return $v * 1099511627776 }
        '^[kK]B$'  { return $v * 1e3 }
        '^MB$'     { return $v * 1e6 }
        '^GB$'     { return $v * 1e9 }
        '^TB$'     { return $v * 1e12 }
        default    { return $v }
    }
}

$InstallCroc = {
    param($AppDir)
    $ErrorActionPreference = 'Stop'
    [Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12
    $ProgressPreference = 'SilentlyContinue'
    $exe = Join-Path $AppDir 'croc.exe'
    if (Test-Path -LiteralPath $exe) { return $exe }
    $arch = if ($env:PROCESSOR_ARCHITECTURE -eq 'ARM64') { 'Windows-ARM64' } else { 'Windows-64bit' }
    $rel = Invoke-RestMethod 'https://api.github.com/repos/schollz/croc/releases/latest' -Headers @{ 'User-Agent' = 'Onde' }
    $asset = $rel.assets | Where-Object { $_.name -like "*_$arch.zip" } | Select-Object -First 1
    if (-not $asset) { throw "Moteur de transfert introuvable pour $arch." }
    $zip = Join-Path $AppDir $asset.name
    Invoke-WebRequest $asset.browser_download_url -OutFile $zip -UseBasicParsing
    $sums = $rel.assets | Where-Object { $_.name -like '*checksums.txt' } | Select-Object -First 1
    if ($sums) {
        $txt = (Invoke-WebRequest $sums.browser_download_url -UseBasicParsing).Content
        if ($txt -is [byte[]]) { $txt = [Text.Encoding]::UTF8.GetString($txt) }
        $line = ($txt -split "`n") | Where-Object { $_ -match [regex]::Escape($asset.name) } | Select-Object -First 1
        if ($line) {
            $expected = ($line.Trim() -split '\s+')[0].ToLower()
            $actual = (Get-FileHash -LiteralPath $zip -Algorithm SHA256).Hash.ToLower()
            if ($expected -ne $actual) { Remove-Item -LiteralPath $zip -Force; throw 'Téléchargement du moteur corrompu (empreinte SHA-256 invalide).' }
        }
    }
    $tmp = Join-Path $AppDir 'extract'
    if (Test-Path -LiteralPath $tmp) { Remove-Item -LiteralPath $tmp -Recurse -Force }
    Expand-Archive -LiteralPath $zip -DestinationPath $tmp -Force
    $found = Get-ChildItem -LiteralPath $tmp -Recurse -Filter 'croc.exe' | Select-Object -First 1
    Move-Item -LiteralPath $found.FullName -Destination $exe -Force
    Remove-Item -LiteralPath $zip, $tmp -Recurse -Force
    return $exe
}
# Inventaire complet : chaque fichier et dossier (cachés, système, vides compris)
$ScanPaths = {
    param([string[]]$paths)
    $ErrorActionPreference = 'Continue'
    $size = 0L; $files = 0; $dirs = 0; $errs = 0
    $lines = New-Object System.Collections.Generic.List[string]
    $bad = New-Object System.Collections.Generic.List[string]
    foreach ($p in $paths) {
        $parent = (Split-Path $p -Parent).TrimEnd('\')
        $root = Get-Item -LiteralPath $p -Force
        $items = New-Object System.Collections.Generic.List[object]
        $items.Add($root)
        if ($root.PSIsContainer) {
            $ev = $null
            foreach ($c in (Get-ChildItem -LiteralPath $p -Recurse -Force -ErrorAction SilentlyContinue -ErrorVariable ev)) { $items.Add($c) }
            if ($ev) { $errs += $ev.Count; foreach ($e in ($ev | Select-Object -First 5)) { $bad.Add([string]$e.TargetObject) } }
        }
        foreach ($it in $items) {
            $f = ''
            if ($it.Attributes -band [IO.FileAttributes]::Hidden) { $f += 'H' }
            if ($it.Attributes -band [IO.FileAttributes]::System) { $f += 'S' }
            if (-not $it.PSIsContainer -and ($it.Attributes -band [IO.FileAttributes]::ReadOnly)) { $f += 'R' }
            $rel = $it.FullName.Substring($parent.Length).TrimStart('\')
            if ($it.PSIsContainer) { $dirs++; $lines.Add("D`t0`t$f`t$rel") }
            else { $files++; $size += $it.Length; $lines.Add("F`t$($it.Length)`t$f`t$rel") }
        }
    }
    return [pscustomobject]@{ Size = $size; Files = $files; Dirs = $dirs; Lines = $lines.ToArray(); Errors = $errs; Bad = $bad.ToArray() }
}

$MakeSound = {
    param($notes, [double]$vol)
    $rate = 44100
    $len = 0.0; foreach ($nt in $notes) { $len = [math]::Max($len, [double]$nt[1] + [double]$nt[2]) }
    $n = [int](($len + 0.05) * $rate)
    $buf = New-Object double[] $n
    foreach ($nt in $notes) {
        $f = [double]$nt[0]; $s = [int]([double]$nt[1] * $rate); $d = [int]([double]$nt[2] * $rate); $dur = [double]$nt[2]
        $w = 2 * [math]::PI * $f / $rate
        for ($i = 0; $i -lt $d -and ($s + $i) -lt $n; $i++) {
            $t = $i / $rate
            $env = [math]::Min(1.0, $i / 300.0) * [math]::Exp(-4.5 * $t / $dur)
            $buf[$s + $i] += $vol * $env * ([math]::Sin($w * $i) + 0.25 * [math]::Sin(2 * $w * $i))
        }
    }
    $ms = New-Object IO.MemoryStream
    $bw = New-Object IO.BinaryWriter($ms)
    $bw.Write([Text.Encoding]::ASCII.GetBytes('RIFF')); $bw.Write([int](36 + $n * 2))
    $bw.Write([Text.Encoding]::ASCII.GetBytes('WAVEfmt ')); $bw.Write([int]16); $bw.Write([int16]1); $bw.Write([int16]1)
    $bw.Write([int]$rate); $bw.Write([int]($rate * 2)); $bw.Write([int16]2); $bw.Write([int16]16)
    $bw.Write([Text.Encoding]::ASCII.GetBytes('data')); $bw.Write([int]($n * 2))
    for ($i = 0; $i -lt $n; $i++) { $v = [math]::Max(-1.0, [math]::Min(1.0, $buf[$i])); $bw.Write([int16]($v * 32000)) }
    $bw.Flush()
    return [pscustomobject]@{ B = $ms.ToArray() }
}

function Start-Croc([string]$arguments, [hashtable]$envVars) {
    $psi = New-Object Diagnostics.ProcessStartInfo
    $psi.FileName = $S.Croc
    $psi.Arguments = $arguments
    $psi.UseShellExecute = $false
    $psi.CreateNoWindow = $true
    $psi.RedirectStandardOutput = $true
    $psi.RedirectStandardError = $true
    $psi.StandardOutputEncoding = [Text.Encoding]::UTF8
    $psi.StandardErrorEncoding = [Text.Encoding]::UTF8
    $psi.WorkingDirectory = $AppDir
    if ($envVars) { foreach ($k in $envVars.Keys) { $psi.EnvironmentVariables[$k] = [string]$envVars[$k] } }
    $p = [Diagnostics.Process]::Start($psi)
    $S.Proc = $p
    $S.Readers = @(
        @{ SR = $p.StandardOutput; Buf = (New-Object char[] 8192); Task = $null; Acc = ''; Done = $false },
        @{ SR = $p.StandardError;  Buf = (New-Object char[] 8192); Task = $null; Acc = ''; Done = $false }
    )
    foreach ($r in $S.Readers) { $r.Task = $r.SR.ReadAsync($r.Buf, 0, $r.Buf.Length) }
    $S.Running = $true
    $S.Done = 0.0; $S.CurName = $null; $S.CurMax = 0.0; $S.CurCounted = $false; $S.LastPct = 0; $S.GotProgress = $false
    Set-Awake $true
}
function Stop-Croc {
    $S.Running = $false
    if ($S.Proc) { try { if (-not $S.Proc.HasExited) { $S.Proc.Kill() } } catch {} }
    $S.Proc = $null
    Set-Awake $false
}
$ESC = [string][char]27
$RxProg = '^(.*?)\s*(\d{1,3})%\s*\|.*?\|\s*\(([\d.]+)/([\d.]+)\s*([kKMGTP]?i?B)(?:,\s*([\d.]+)\s*([kKMGTP]?i?B)/s)?\)'
$RxTotal = '^(Sending|Receiving|Accept)\s.*\(([\d.]+)\s*([kKMGTP]?i?B)\)'
function Read-Croc {
    foreach ($r in $S.Readers) {
        while (-not $r.Done -and $r.Task.IsCompleted) {
            $n = 0
            try { $n = $r.Task.Result } catch { $n = 0 }
            if ($n -le 0) { $r.Done = $true; break }
            $r.Acc += [string]::new($r.Buf, 0, $n)
            $r.Task = $r.SR.ReadAsync($r.Buf, 0, $r.Buf.Length)
        }
        if ($r.Acc.IndexOfAny([char[]]"`r`n") -ge 0) {
            $parts = $r.Acc -split "[`r`n]"
            $r.Acc = $parts[-1]
            for ($i = 0; $i -lt $parts.Count - 1; $i++) { Read-Line $parts[$i] }
        }
        if ($r.Done -and $r.Acc) { Read-Line $r.Acc; $r.Acc = '' }
    }
}
function Read-Line([string]$l) {
    if ($S.View -eq 'VError') { return }
    $l = ($l -replace ($ESC + '\[[0-9;?]*[A-Za-z]'), '').Trim()
    if (-not $l) { return }
    if ($l -notmatch '%\s*\|') { $S.Lines.Add($l); if ($S.Lines.Count -gt 12) { $S.Lines.RemoveAt(0) } }
    if ($l -match $RxTotal) {
        $S.Total = ConvertTo-Bytes $Matches[2] $Matches[3]
        if ($Mode -eq 'recv') { Test-Space }
        return
    }
    if ($l -match $RxProg) {
        $name = $Matches[1].Trim(); $pct = [int]$Matches[2]
        $cur = ConvertTo-Bytes $Matches[3] $Matches[5]; $max = ConvertTo-Bytes $Matches[4] $Matches[5]
        if ($Matches[6]) { $S.Speed = ConvertTo-Bytes $Matches[6] $Matches[7] }
        if ($name -match '(?i)hashing|hash\b|calcul') {
            $S.HashSeen = $true; $S.LastHash = Get-Date
            if (-not $S.GotProgress) {
                if ($name -ne $S.HashName) { if ($S.HashName) { $S.HashDone += $S.HashMax }; $S.HashName = $name }
                $S.HashMax = $max
                $tot = if ($S.Size -gt 0) { $S.Size } else { $max }
                $hp = if ($tot -gt 0) { [int][math]::Min(100, [math]::Floor(100 * ($S.HashDone + $cur) / $tot)) } else { $pct }
                $S.HashPct = $hp
                if ($hp -lt 100) {
                    if ($S.View -eq 'VWait') { $ui.WaitState.Text = "Préparation du dossier… $hp %"; $ui.WaitInfo.Text = 'Le partage se débloquera à 100 %. Patiente quelques instants…' }
                    elseif ($S.View -eq 'VPrep') { $ui.PrepTitle.Text = "Préparation… $hp %"; $ui.PrepSub.Text = 'Vérification des fichiers…' }
                }
            } elseif ($S.View -eq 'VRun') { $ui.CurFile.Text = 'Vérification finale des fichiers…'; if ($S.Speed -gt 0) { $ui.StSpeed.Text = (Format-Size $S.Speed) + '/s' } }
            return
        }
        if ($name -ne $S.CurName -or $pct -lt ($S.LastPct - 5)) {
            if ($null -ne $S.CurName -and -not $S.CurCounted) { $S.Done += $S.CurMax }
            $S.CurName = $name; $S.CurCounted = $false
        }
        $S.CurMax = $max; $S.LastPct = $pct
        if ($pct -ge 100 -and -not $S.CurCounted) { $S.Done += $max; $S.CurCounted = $true }
        $S.Overall = $S.Done + $(if ($S.CurCounted) { 0 } else { $cur })
        if (-not $S.GotProgress) { $S.GotProgress = $true; $S.Retries = 0; Show-Run }
        $total = if ($S.Total -gt 0) { $S.Total } else { $max }
        $shown = if ($total -gt 0) { [math]::Min($S.Overall, $total) } else { $S.Overall }
        $S.TargetPct = if ($total -gt 0) { [math]::Min(99.9, 100 * $shown / $total) } else { $pct }
        $ui.StDone.Text = (Format-Size $shown) + ' / ' + (Format-Size $total)
        if ($S.Speed -gt 0) {
            $ui.StSpeed.Text = (Format-Size $S.Speed) + '/s'
            if ($total -gt $shown) { $ui.StEta.Text = Format-Dur (($total - $shown) / $S.Speed) } else { $ui.StEta.Text = 'presque fini' }
        }
        if ($name) { $ui.CurFile.Text = $name }
    }
}

        }
    } catch {}
}

function New-Code {
    $rng = [Security.Cryptography.RandomNumberGenerator]::Create()
    $alpha = 'abcdefghjkmnpqrstuvwxyz23456789'
    $mk = { param($n, $chars) $b = New-Object byte[] $n; $rng.GetBytes($b); -join ($b | ForEach-Object { $chars[$_ % $chars.Length] }) }
    return ('{0}-{1}-{2}-{3}' -f (& $mk 4 '0123456789'), (& $mk 5 $alpha), (& $mk 5 $alpha), (& $mk 5 $alpha))
}

function Get-CodeFrom([string]$t) {
    if (-not $t) { return '' }
    $m = [regex]::Match($t, '\d{4}(-[a-z0-9]{5}){3}')
    if ($m.Success) { return $m.Value }
    return ($t -replace '^(?i)\s*code d.acc[eè]s\s*:\s*', '').Trim()
}
function Get-KnownDirs {
    $dl = $null
    try { $dl = (New-Object -ComObject Shell.Application).Namespace('shell:Downloads').Self.Path } catch {}
    if (-not $dl) { try { $v = (Get-ItemProperty 'HKCU:\Software\Microsoft\Windows\CurrentVersion\Explorer\User Shell Folders' -ErrorAction Stop).'{374DE290-123F-4565-9164-39C4925E467B}'; if ($v) { $dl = [Environment]::ExpandEnvironmentVariables($v) } } catch {} }
    if (-not $dl) { $dl = Join-Path $env:USERPROFILE 'Downloads' }
    return @(
        @{ L = 'Téléchargements'; P = $dl },
        @{ L = 'Bureau'; P = [Environment]::GetFolderPath('Desktop') },
        @{ L = 'Documents'; P = [Environment]::GetFolderPath('MyDocuments') },
        @{ L = 'Images'; P = [Environment]::GetFolderPath('MyPictures') },
        @{ L = 'Vidéos'; P = [Environment]::GetFolderPath('MyVideos') }
    ) | Where-Object { $_.P }
}
function Get-DestLabel {
    foreach ($k in (Get-KnownDirs)) { if ($S.Dest -and $k.P.TrimEnd('\') -ieq $S.Dest.TrimEnd('\')) { return $k.L } }

}

    } catch {}
    return (Get-KnownDirs)[0].P
}

function Get-CodeHash([string]$c) {
    $sha = [Security.Cryptography.SHA256]::Create()
    return (-join ($sha.ComputeHash([Text.Encoding]::UTF8.GetBytes('onde|' + $c)) | ForEach-Object { $_.ToString('x2') }))
}

    if (-not $AppDir) { return }

function Recv-Finished {
    $verify = ''; $ok = $true; $root = $null
    $src = if ($S.Stage -and (Test-Path -LiteralPath $S.Stage)) { $S.Stage } else { $S.Dest }
    $attrs = New-Object System.Collections.Generic.List[object]
    try {
        $mf = Get-ChildItem -LiteralPath $src -Filter '*.sillage' -File -Force | Sort-Object LastWriteTime -Descending | Select-Object -First 1
        if ($mf) {
            $nF = 0; $nD = 0; $missing = New-Object System.Collections.Generic.List[string]
            foreach ($line in [IO.File]::ReadAllLines($mf.FullName, [Text.Encoding]::UTF8)) {
                $parts = $line -split "`t"
                if ($parts.Count -lt 4) { continue }
                $rel = ($parts[3..($parts.Count - 1)] -join "`t")
                $segs = $rel.Split([char]92, [char]47)
                if (($segs -contains '..') -or [IO.Path]::IsPathRooted($rel)) { continue }
                if (-not $root) { $root = $segs[0] }
                $sp = Join-Path $src $rel
                if ($parts[0] -eq 'D') {
                    $nD++
                    if (-not (Test-Path -LiteralPath $sp)) { try { New-Item -ItemType Directory -Force -Path $sp | Out-Null } catch { $missing.Add($rel) } }
                } elseif ($parts[0] -eq 'F') {
                    $nF++
                    $fi = $null; try { $fi = Get-Item -LiteralPath $sp -Force -ErrorAction Stop } catch {}
                    if (-not $fi -or $fi.Length -ne [int64]$parts[1]) { $missing.Add($rel) }
                }
                if ($parts[2]) { $attrs.Add(@($rel, $parts[2])) }
            }
            Remove-Item -LiteralPath $mf.FullName -Force
            if ($nF -gt 0 -or $nD -gt 0) {
                if ($missing.Count -eq 0) { $verify = ('✓  Vérifié : {0:N0} fichiers et {1:N0} dossiers, rien ne manque.' -f $nF, $nD) }
                else { $ok = $false; $ex = ($missing | Select-Object -First 2) -join ', '; $verify = ('Transfert incomplet ({0} élément(s) manquant(s), ex. {1}). Rien n''a été déposé dans ton dossier ; demande un nouvel envoi.' -f $missing.Count, $ex) }
            }
        }
    } catch {}
    # Rien n'est posé dans le dossier final tant que le transfert n'est pas complet et vérifié
    if ($ok -and $S.Stage -and (Test-Path -LiteralPath $S.Stage)) {
        try {
            foreach ($item in (Get-ChildItem -LiteralPath $S.Stage -Force)) {
                if ($item.Extension -eq '.sillage') { continue }
                $target = Join-Path $S.Dest $item.Name
                if (Test-Path -LiteralPath $target) { Remove-Item -LiteralPath $target -Recurse -Force -ErrorAction SilentlyContinue }
                Move-Item -LiteralPath $item.FullName -Destination $target -Force
            }
        } catch {}
        for ($i = $attrs.Count - 1; $i -ge 0; $i--) {
            $p = Join-Path $S.Dest $attrs[$i][0]; $fl = $attrs[$i][1]
            try {
                $item = Get-Item -LiteralPath $p -Force -ErrorAction Stop
                $a = $item.Attributes
                if ($fl -match 'H') { $a = $a -bor [IO.FileAttributes]::Hidden }
                if ($fl -match 'S') { $a = $a -bor [IO.FileAttributes]::System }
                if ($fl -match 'R') { $a = $a -bor [IO.FileAttributes]::ReadOnly }
                $item.Attributes = $a
            } catch {}
        }
    }
    $hadStage = [bool]($S.Stage)
    Clear-RecvTraces
    $S.Opened = if ($root) { Join-Path $S.Dest $root } else { $S.Dest }
    $size = if ($S.Total -gt 0) { ' (' + (Format-Size $S.Total) + ')' } else { '' }
    if ($ok) {
        Show-Done 'Dossier reçu' "Tout est arrivé$size dans « $(Get-DestLabel) », dans son format d'origine." $verify $true 'Ouvrir le dossier'
        Open-Received
    } else {
        Show-Done 'Transfert incomplet' "Le transfert n'est pas allé au bout. Rien n'a été ajouté à ton dossier, donc aucun fichier abîmé. Demande un nouvel envoi." $verify $false 'Ouvrir le dossier'
    }
}
function Open-Received {
    try {
        if ($S.Opened -and (Test-Path -LiteralPath $S.Opened) -and $S.Opened -ne $S.Dest) { Start-Process explorer.exe ('/select,"' + $S.Opened + '"') }
        else { Start-Process explorer.exe ('"' + $S.Dest + '"') }
    } catch {}
}

# ---------------------------------------------------------------- particules et animation (~30 images/s)
$PBrushes = @((New-Brush 240 225 29 46), (New-Brush 230 255 92 104), (New-Brush 230 255 255 255), (New-Brush 200 255 180 186))
function Update-Fx {
    $S.Tick++
    # barre lissée
    $S.ShownPct += ($S.TargetPct - $S.ShownPct) * 0.12
    if ($S.View -eq 'VRun') {
        $tw = $ui.BarTrack.ActualWidth; if ($tw -le 0) { $tw = 468 }
        $w = [math]::Max(0, $tw * $S.ShownPct / 100)
        $ui.BarFill.Width = $w
        $ui.Pct.Text = ('{0:N0} %' -f [math]::Floor($S.ShownPct))
        # émission légère depuis la tête de la barre
        if ($w -gt 6 -and $S.Particles.Count -lt 36 -and ($S.Tick % 2 -eq 0) -and $S.Running) {
            $e = New-Object Windows.Shapes.Ellipse
            $sz = 2 + $S.Rand.NextDouble() * 2.5
            $e.Width = $sz; $e.Height = $sz
            $e.Fill = $PBrushes[$S.Rand.Next(0, 4)]
            $e.IsHitTestVisible = $false
            [void]$ui.Fx.Children.Add($e)
            [void]$S.Particles.Add(@{ E = $e; X = $w - 3; Y = 20 - $sz / 2; VX = -0.4 - $S.Rand.NextDouble() * 1.2; VY = -0.6 + $S.Rand.NextDouble() * 1.2; L = 1.0; D = 0.02 + $S.Rand.NextDouble() * 0.025 })
        }
    }
    for ($i = $S.Particles.Count - 1; $i -ge 0; $i--) {
        $p = $S.Particles[$i]
        $p.X += $p.VX; $p.Y += $p.VY; $p.VY *= 0.96; $p.L -= $p.D
        if ($p.L -le 0) { [void]$ui.Fx.Children.Remove($p.E); $S.Particles.RemoveAt($i); continue }
        [Windows.Controls.Canvas]::SetLeft($p.E, $p.X); [Windows.Controls.Canvas]::SetTop($p.E, $p.Y)
        $p.E.Opacity = $p.L
    }
    # respiration du point « en attente »
    if ($LiveDot) { $LiveDot.Opacity = 0.55 + 0.45 * [math]::Abs([math]::Sin($S.Tick / 12.0)) }
    if ($S.View -eq 'VWait') { $ui.WaitDot.Opacity = 0.45 + 0.55 * [math]::Abs([math]::Sin($S.Tick / 9.0)) }
    if ($S.View -eq 'VIdle') { $ui.DropFloat.Y = 3 * [math]::Sin($S.Tick / 14.0) }
}

# ---------------------------------------------------------------- boucle de contrôle (4 fois par seconde)
function Update-Logic {
    if ($S.Flash.Count) { Reset-Flash }
    if ($Mode -eq 'send' -and $S.View -eq 'VWait' -and -not $S.SendReady) {
        if ($S.GotProgress -or ($S.HashSeen -and $S.HashPct -ge 100) -or ($S.HashSeen -and $S.LastHash -and ((Get-Date) - $S.LastHash).TotalSeconds -gt 2.5) -or (-not $S.HashSeen -and $S.SendStart -and ((Get-Date) - $S.SendStart).TotalSeconds -gt 5)) {
            Set-SendReady
        }
    }
    if ($Mode -eq 'recv' -and $S.View -eq 'VPrep' -and $S.Running -and $S.StartTime -and -not $S.GotProgress -and ((Get-Date) - $S.StartTime).TotalSeconds -gt 7) {
        $ui.PrepTitle.Text = "En attente de l'expéditeur…"
        $ui.PrepSub.Text = "Le transfert démarre dès que l'expéditeur a Onde ouvert sur son PC. Destination : « $(Get-DestLabel) »."
    }
    if ($S.Bg.ContainsKey('croc')) {
        $r = Receive-Bg 'croc'
        if ($r) {
            if ($r.Ok) { $S.Croc = [string]$r.Value }
            else { Show-Error ("Impossible d'installer le module de transfert. Vérifie ta connexion internet.`n" + $r.Error); return }
            if ($Mode -eq 'recv' -and $S.View -eq 'VPrep') { Start-RecvCroc }
        }
    }
    if ($S.Bg.ContainsKey('scan')) {
        $r = Receive-Bg 'scan'
        if ($r) { if ($r.Ok) { $S.Scan = $r.Value } else { Show-Error ("Impossible de lire le dossier.`n" + $r.Error); return } }
    }
    foreach ($k in @('snd1', 'snd2')) {
        if ($S.Bg.ContainsKey($k)) { $r = Receive-Bg $k; if ($r -and $r.Ok -and $r.Value) { if ($k -eq 'snd1') { $S.SndDrop = $r.Value.B } else { $S.SndDone = $r.Value.B } } }
    }
    if ($Mode -eq 'send' -and $S.View -eq 'VPrep' -and $S.Scan -and $S.Croc -and -not $S.Code) { Send-Ready }

    if ($S.RetryAt -and (Get-Date) -ge $S.RetryAt) {
        $S.RetryAt = $null
        if ($Mode -eq 'send') { Start-SendCroc } else { Start-RecvCroc }
    }
    if ($S.Running -and $S.Proc) {
        Read-Croc
        if ($S.Proc -and $S.Proc.HasExited -and -not ($S.Readers | Where-Object { -not $_.Done })) {
            $code = $S.Proc.ExitCode
            $S.Running = $false; $S.Proc = $null; Set-Awake $false
            if ($S.View -eq 'VError') { return }
            if ($code -eq 0) {
                $S.TargetPct = 100
                if ($Mode -eq 'send') { Send-Finished } else { Recv-Finished }
            } else {
                $S.Retries++
                $max = if ($Mode -eq 'send') { 300 } else { 45 }
                if ($S.Retries -gt $max) {
                    $last = ($S.Lines | Select-Object -Last 2) -join "`n"
                    if ($Mode -eq 'send') { Show-Error "La connexion n'a pas pu être rétablie.`n$last" }
                    else { Show-Error "Ce code n'est plus actif : soit le dossier a déjà été reçu (un code ne sert qu'une fois), soit l'expéditeur a fermé Onde. Demande-lui de relancer l'envoi, puis clique sur Réessayer." }
                    return
                }
                $S.RetryAt = (Get-Date).AddSeconds(4)
                if ($S.View -eq 'VRun') {
                    $ui.RunSub.Text = 'Connexion perdue : reprise automatique là où le transfert s''est arrêté…'
                    $ui.RunSub.Foreground = New-Brush 255 251 191 36
                } elseif ($S.View -eq 'VWait') {
                    $ui.WaitState.Text = 'En attente de ton client… (reconnexion)'
                }
            }
        }
    }
}

# ---------------------------------------------------------------- logo, icône et raccourci
function Get-LogoBitmap {
    $logo = [Windows.Markup.XamlReader]::Parse($LogoXaml)
    $sz = New-Object Windows.Size(256, 256)
    $logo.Measure($sz); $logo.Arrange((New-Object Windows.Rect($sz))); $logo.UpdateLayout()
    $rtb = New-Object Windows.Media.Imaging.RenderTargetBitmap(256, 256, 96, 96, [Windows.Media.PixelFormats]::Pbgra32)
    $rtb.Render($logo)
    return $rtb
}
function Install-App($bmp) {
    try {
        $ico = Join-Path $AppDir 'onde-v5.ico'
        if ($bmp -and -not (Test-Path -LiteralPath $ico)) {
            $enc = New-Object Windows.Media.Imaging.PngBitmapEncoder
            $enc.Frames.Add([Windows.Media.Imaging.BitmapFrame]::Create($bmp))
            $pngMs = New-Object IO.MemoryStream; $enc.Save($pngMs); $png = $pngMs.ToArray()
            $fs = [IO.File]::Create($ico); $bw = New-Object IO.BinaryWriter($fs)
            $bw.Write([int16]0); $bw.Write([int16]1); $bw.Write([int16]1)
            $bw.Write([byte]0); $bw.Write([byte]0); $bw.Write([byte]0); $bw.Write([byte]0)
            $bw.Write([int16]1); $bw.Write([int16]32); $bw.Write([int]$png.Length); $bw.Write([int]22)
            $bw.Write($png); $bw.Close()
        }
        $cmd = Join-Path $AppDir 'Onde.cmd'
        $content = "@echo off`r`nset `"TS_ARGS=%*`"`r`nset `"TS_MODE=$Mode`"`r`nstart `"`" powershell -NoProfile -ExecutionPolicy Bypass -Command `"[Net.ServicePointManager]::SecurityProtocol='Tls12'; iex (irm ${Site}app.txt)`"`r`n"
        if (-not (Test-Path -LiteralPath $cmd)) { [IO.File]::WriteAllText($cmd, $content, (New-Object Text.ASCIIEncoding)) }
        $desk = [Environment]::GetFolderPath('Desktop')
        $lnkPath = Join-Path $desk 'Onde.lnk'
        $ws = New-Object -ComObject WScript.Shell
        $lnk = $ws.CreateShortcut($lnkPath)
        $lnk.TargetPath = $cmd
        $lnk.WorkingDirectory = $AppDir
        $lnk.WindowStyle = 7
        if (Test-Path -LiteralPath $ico) { $lnk.IconLocation = "$ico,0" }
        $lnk.Description = 'Onde : envoyer et recevoir des dossiers chiffrés'
        $lnk.Save()
        foreach ($o in @('Transfert sécurisé.lnk', 'Sillage.lnk')) { $old = Join-Path $desk $o; if (Test-Path -LiteralPath $old) { Remove-Item -LiteralPath $old -Force } }
    } catch {}
}

function Flash-Copied($btn, [string]$label) {
    if ($null -eq $btn.Tag) { $btn.Tag = [string]$btn.Content }
    $btn.Content = $label
    $btn.Background = New-Brush 255 34 197 94
    $btn.Foreground = New-Brush 255 255 255 255
    $btn.RenderTransformOrigin = New-Object Windows.Point(0.5, 0.5)
    $st = New-Object Windows.Media.ScaleTransform(1.0, 1.0)
    $btn.RenderTransform = $st
    $ease = New-Object Windows.Media.Animation.BackEase; $ease.Amplitude = 0.6; $ease.EasingMode = 'EaseOut'
    $a = New-Object Windows.Media.Animation.DoubleAnimation(0.92, 1.0, (New-Object Windows.Duration ([TimeSpan]::FromMilliseconds(380))))
    $a.EasingFunction = $ease
    $st.BeginAnimation([Windows.Media.ScaleTransform]::ScaleXProperty, $a)
    $st.BeginAnimation([Windows.Media.ScaleTransform]::ScaleYProperty, $a)
    $S.Flash[$btn.Name] = (Get-Date).AddSeconds(2.8)

}
function Reset-Flash {
    foreach ($k in @($S.Flash.Keys)) {
        if ((Get-Date) -ge $S.Flash[$k]) {
            $b = $ui[$k]
            if ($b) {
                $b.Content = $b.Tag
                $b.ClearValue([Windows.Controls.Control]::BackgroundProperty)
                $b.ClearValue([Windows.Controls.Control]::ForegroundProperty)
            }
            $S.Flash.Remove($k)
        }
    }
}

function Fill-Browse([string]$dir) {
    $ui.BrowseList.Items.Clear()
    if ($dir -eq '::drives') {
        $Browse.Cur = '::drives'; $ui.BrowsePath.Text = 'Ce PC'
        foreach ($d in [IO.DriveInfo]::GetDrives()) {
            if ($d.IsReady) {
                $n = $d.Name.TrimEnd('\')
                $lbl = if ($d.VolumeLabel) { $d.VolumeLabel + ' (' + $n + ')' } else { 'Disque (' + $n + ')' }
                Add-BrowseItem $lbl $d.RootDirectory.FullName 'drive'
            }
        }
        return
    }
    $Browse.Cur = $dir; $ui.BrowsePath.Text = $dir
    $parent = [IO.Path]::GetDirectoryName($dir.TrimEnd('\'))
    if ($parent) { Add-BrowseItem 'Dossier parent' $parent 'up' } else { Add-BrowseItem 'Ce PC' '::drives' 'up' }
    try {
        $subs = Get-ChildItem -LiteralPath $dir -Directory -Force -ErrorAction SilentlyContinue | Where-Object { -not ($_.Attributes -band [IO.FileAttributes]::System) -and -not ($_.Attributes -band [IO.FileAttributes]::Hidden) } | Sort-Object Name
        foreach ($s in $subs) { Add-BrowseItem $s.Name $s.FullName 'dir' }
    } catch {}
    if ($ui.BrowseList.Items.Count) { $ui.BrowseList.ScrollIntoView($ui.BrowseList.Items[0]) }
}
