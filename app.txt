# ---------------------------------------------------------------- utilitaires
function Format-Size([double]$b) {
    if ($b -ge 1e12) { return ('{0:N2} To' -f ($b / 1e12)) }
    if ($b -ge 1e9)  { return ('{0:N1} Go' -f ($b / 1e9)) }
    if ($b -ge 1e6)  { return ('{0:N1} Mo' -f ($b / 1e6)) }
    if ($b -ge 1e3)  { return ('{0:N0} Ko' -f ($b / 1e3)) }
    return ('{0:N0} o' -f $b)
}
function Format-Dur([double]$s) {
    if ($s -lt 0 -or [double]::IsInfinity($s) -or [double]::IsNaN($s)) { return '' }
    if ($s -lt 60) { return ('{0:N0} s' -f $s) }
    if ($s -lt 3600) { return ('{0:N0} min' -f [math]::Ceiling($s / 60)) }
    $h = [math]::Floor($s / 3600); $m = [math]::Floor(($s % 3600) / 60)
    return ('{0} h {1:00}' -f $h, $m)
}
function ConvertTo-Bytes([string]$n, [string]$u) {
    $v = [double]::Parse($n, [Globalization.CultureInfo]::InvariantCulture)
    switch -regex ($u) {
        '^[kK]iB$' { return $v * 1024 }
        '^MiB$'    { return $v * 1048576 }
        '^GiB$'    { return $v * 1073741824 }
        '^TiB$'    { return $v * 1099511627776 }
        '^[kK]B$'  { return $v * 1e3 }
        '^MB$'     { return $v * 1e6 }
        '^GB$'     { return $v * 1e9 }
        '^TB$'     { return $v * 1e12 }
        default    { return $v }
    }
}
function Set-Awake([bool]$on) {
    try { if ($on) { [void][TSApp.Native]::SetThreadExecutionState([uint32]2147483649) } else { [void][TSApp.Native]::SetThreadExecutionState([uint32]2147483648) } } catch {}
}
function Copy-Text([string]$t) {
    for ($i = 0; $i -lt 5; $i++) { try { [Windows.Forms.Clipboard]::SetText($t); return $true } catch { Start-Sleep -Milliseconds 120 } }
    return $false
}
function Start-Bg([string]$key, [scriptblock]$sb, [object[]]$argList) {
    $ps = [powershell]::Create()
    [void]$ps.AddScript($sb.ToString())
    foreach ($a in $argList) { [void]$ps.AddArgument($a) }
    $S.Bg[$key] = @{ PS = $ps; H = $ps.BeginInvoke() }
}
function Receive-Bg([string]$key) {
    # renvoie $null si pas fini ; sinon @{ Ok; Value; Error }
    $bg = $S.Bg[$key]
    if (-not $bg -or -not $bg.H.IsCompleted) { return $null }
    $S.Bg.Remove($key)
    try {
        $out = $bg.PS.EndInvoke($bg.H)
        if ($bg.PS.HadErrors -and $out.Count -eq 0) { $msg = ($bg.PS.Streams.Error | Select-Object -First 1 | Out-String).Trim(); return @{ Ok = $false; Error = $msg } }
        return @{ Ok = $true; Value = ($out | Select-Object -Last 1) }
    } catch { return @{ Ok = $false; Error = $_.Exception.InnerException.Message + $_.Exception.Message } }
    finally { $bg.PS.Dispose() }
}

$InstallCroc = {
    param($AppDir)
    $ErrorActionPreference = 'Stop'
    [Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12
    $ProgressPreference = 'SilentlyContinue'
    $exe = Join-Path $AppDir 'croc.exe'
    if (Test-Path -LiteralPath $exe) { return $exe }
    $arch = if ($env:PROCESSOR_ARCHITECTURE -eq 'ARM64') { 'Windows-ARM64' } else { 'Windows-64bit' }
    $rel = Invoke-RestMethod 'https://api.github.com/repos/schollz/croc/releases/latest' -Headers @{ 'User-Agent' = 'TransfertSecurise' }
    $asset = $rel.assets | Where-Object { $_.name -like "*_$arch.zip" } | Select-Object -First 1
    if (-not $asset) { throw "Moteur de transfert introuvable pour $arch." }
    $zip = Join-Path $AppDir $asset.name
    Invoke-WebRequest $asset.browser_download_url -OutFile $zip -UseBasicParsing
    $sums = $rel.assets | Where-Object { $_.name -like '*checksums.txt' } | Select-Object -First 1
    if ($sums) {
        $txt = (Invoke-WebRequest $sums.browser_download_url -UseBasicParsing).Content
        if ($txt -is [byte[]]) { $txt = [Text.Encoding]::UTF8.GetString($txt) }
        $line = ($txt -split "`n") | Where-Object { $_ -match [regex]::Escape($asset.name) } | Select-Object -First 1
        if ($line) {
            $expected = ($line.Trim() -split '\s+')[0].ToLower()
            $actual = (Get-FileHash -LiteralPath $zip -Algorithm SHA256).Hash.ToLower()
            if ($expected -ne $actual) { Remove-Item -LiteralPath $zip -Force; throw 'Téléchargement du moteur corrompu (empreinte SHA-256 invalide).' }
        }
    }
    $tmp = Join-Path $AppDir 'extract'
    if (Test-Path -LiteralPath $tmp) { Remove-Item -LiteralPath $tmp -Recurse -Force }
    Expand-Archive -LiteralPath $zip -DestinationPath $tmp -Force
    $found = Get-ChildItem -LiteralPath $tmp -Recurse -Filter 'croc.exe' | Select-Object -First 1
    Move-Item -LiteralPath $found.FullName -Destination $exe -Force
    Remove-Item -LiteralPath $zip, $tmp -Recurse -Force
    return $exe
}
$ScanPaths = {
    param([string[]]$paths)
    $ErrorActionPreference = 'Continue'
    $size = 0L; $files = 0; $errs = 0
    $lines = New-Object System.Collections.Generic.List[string]
    foreach ($p in $paths) {
        $parent = (Split-Path $p -Parent).TrimEnd('\')
        $root = Get-Item -LiteralPath $p -Force
        $items = New-Object System.Collections.Generic.List[object]
        $items.Add($root)
        if ($root.PSIsContainer) {
            $ev = $null
            foreach ($c in (Get-ChildItem -LiteralPath $p -Recurse -Force -ErrorAction SilentlyContinue -ErrorVariable ev)) { $items.Add($c) }
            if ($ev) { $errs += $ev.Count }
        }
        foreach ($it in $items) {
            if (-not $it.PSIsContainer) { $files++; $size += $it.Length }
            $f = ''
            if ($it.Attributes -band [IO.FileAttributes]::Hidden) { $f += 'H' }
            if ($it.Attributes -band [IO.FileAttributes]::System) { $f += 'S' }
            if (-not $it.PSIsContainer -and ($it.Attributes -band [IO.FileAttributes]::ReadOnly)) { $f += 'R' }
            if ($f) { $lines.Add($f + "`t" + $it.FullName.Substring($parent.Length).TrimStart('\')) }
        }
    }
    return [pscustomobject]@{ Size = $size; Files = $files; Attr = $lines.ToArray(); Errors = $errs }
}

# ---------------------------------------------------------------- moteur (croc)
function Start-Croc([string]$arguments, [hashtable]$envVars) {
    $psi = New-Object Diagnostics.ProcessStartInfo
    $psi.FileName = $S.Croc
    $psi.Arguments = $arguments
    $psi.UseShellExecute = $false
    $psi.CreateNoWindow = $true
    $psi.RedirectStandardOutput = $true
    $psi.RedirectStandardError = $true
    $psi.StandardOutputEncoding = [Text.Encoding]::UTF8
    $psi.StandardErrorEncoding = [Text.Encoding]::UTF8
    $psi.WorkingDirectory = $AppDir
    if ($envVars) { foreach ($k in $envVars.Keys) { $psi.EnvironmentVariables[$k] = [string]$envVars[$k] } }
    $p = [Diagnostics.Process]::Start($psi)
    $S.Proc = $p
    $S.Readers = @(
        @{ SR = $p.StandardOutput; Buf = (New-Object char[] 8192); Task = $null; Acc = ''; Done = $false },
        @{ SR = $p.StandardError;  Buf = (New-Object char[] 8192); Task = $null; Acc = ''; Done = $false }
    )
    foreach ($r in $S.Readers) { $r.Task = $r.SR.ReadAsync($r.Buf, 0, $r.Buf.Length) }
    $S.Running = $true
    $S.Done = 0.0; $S.CurName = $null; $S.CurMax = 0.0; $S.CurCounted = $false; $S.LastPct = 0; $S.GotProgress = $false
    Set-Awake $true
}
function Stop-Croc {
    $S.Running = $false
    if ($S.Proc) { try { if (-not $S.Proc.HasExited) { $S.Proc.Kill() } } catch {} }
    $S.Proc = $null
    Set-Awake $false
}
$ESC = [string][char]27
function Read-Croc {
    foreach ($r in $S.Readers) {
        while (-not $r.Done -and $r.Task.IsCompleted) {
            $n = 0
            try { $n = $r.Task.Result } catch { $n = 0 }
            if ($n -le 0) { $r.Done = $true; break }
            $r.Acc += [string]::new($r.Buf, 0, $n)
            $r.Task = $r.SR.ReadAsync($r.Buf, 0, $r.Buf.Length)
        }
        if ($r.Acc.IndexOfAny([char[]]"`r`n") -ge 0) {
            $parts = $r.Acc -split "[`r`n]"
            $r.Acc = $parts[-1]
            for ($i = 0; $i -lt $parts.Count - 1; $i++) { Read-Line $parts[$i] }
        }
        if ($r.Done -and $r.Acc) { Read-Line $r.Acc; $r.Acc = '' }
    }
}
$RxProg = '^(.*?)\s*(\d{1,3})%\s*\|.*?\|\s*\(([\d.]+)/([\d.]+)\s*([kKMGTP]?i?B)(?:,\s*([\d.]+)\s*([kKMGTP]?i?B)/s)?\)'
$RxTotal = '^(Sending|Receiving|Accept)\s.*\(([\d.]+)\s*([kKMGTP]?i?B)\)'
function Read-Line([string]$l) {
    if ($S.View -eq 'error') { return }
    $l = ($l -replace ($ESC + '\[[0-9;?]*[A-Za-z]'), '').Trim()
    if (-not $l) { return }
    if ($l -notmatch '%\s*\|') { $S.Lines.Add($l); if ($S.Lines.Count -gt 12) { $S.Lines.RemoveAt(0) } }
    if ($l -match $RxTotal) {
        $S.Total = ConvertTo-Bytes $Matches[2] $Matches[3]
        if ($Mode -eq 'recv') { Test-Space }
        return
    }
    if ($l -match $RxProg) {
        $name = $Matches[1].Trim(); $pct = [int]$Matches[2]
        $cur = ConvertTo-Bytes $Matches[3] $Matches[5]; $max = ConvertTo-Bytes $Matches[4] $Matches[5]
        if ($Matches[6]) { $S.Speed = ConvertTo-Bytes $Matches[6] $Matches[7] }
        if ($name -ne $S.CurName -or $pct -lt ($S.LastPct - 5)) {
            if ($null -ne $S.CurName -and -not $S.CurCounted) { $S.Done += $S.CurMax }
            $S.CurName = $name; $S.CurCounted = $false
        }
        $S.CurMax = $max; $S.LastPct = $pct
        if ($pct -ge 100 -and -not $S.CurCounted) { $S.Done += $max; $S.CurCounted = $true }
        $S.Overall = $S.Done + $(if ($S.CurCounted) { 0 } else { $cur })
        if (-not $S.GotProgress) {
            $S.GotProgress = $true; $S.Retries = 0
            if ($Mode -eq 'send') { Set-View 'run' 'Envoi en cours' "Ton client télécharge « $($S.Name) ». Laisse cette fenêtre ouverte." }
            else { Set-View 'run' 'Réception en cours' "Enregistrement dans : $($S.Dest)`nLaisse cette fenêtre ouverte jusqu'à la fin." }
            Set-Buttons '' 'Annuler'
        }
        $total = if ($S.Total -gt 0) { $S.Total } else { $max }
        $gpct = if ($total -gt 0) { 100 * $S.Overall / $total } else { $pct }
        $gpct = [math]::Min(99.9, $gpct)
        Set-Fill $gpct
        $eta = if ($S.Speed -gt 0) { Format-Dur (($total - $S.Overall) / $S.Speed) } else { '' }
        $parts = @(('{0:N0} %' -f [math]::Floor($gpct)), ((Format-Size $S.Overall) + ' / ' + (Format-Size $total)))
        if ($S.Speed -gt 0) { $parts += ((Format-Size $S.Speed) + '/s') }
        if ($eta) { $parts += ('reste ' + $eta) }
        $lblStat.Text = $parts -join '   ·   '
        $lblFile.Text = $name
    }
}
function Test-Space {
    try {
        $drive = (Get-Item -LiteralPath $S.Dest).PSDrive
        if ($drive -and $drive.Free -and $S.Total -gt ($drive.Free + 1e8)) {
            $need = Format-Size $S.Total; $free = Format-Size $drive.Free
            Show-Error "Pas assez de place sur le disque $($drive.Name): pour recevoir ce dossier.`nIl faut $need, il reste $free. Libère de la place puis clique sur Réessayer."
        }
    } catch {}
}

function New-Code {
    $rng = [Security.Cryptography.RandomNumberGenerator]::Create()
    $alpha = 'abcdefghjkmnpqrstuvwxyz23456789'
    $mk = { param($n, $chars) $b = New-Object byte[] $n; $rng.GetBytes($b); -join ($b | ForEach-Object { $chars[$_ % $chars.Length] }) }
    return ('{0}-{1}-{2}-{3}' -f (& $mk 4 '0123456789'), (& $mk 5 $alpha), (& $mk 5 $alpha), (& $mk 5 $alpha))
}
function Start-Send([string[]]$paths) {
    $paths = @($paths | Where-Object { $_ -and (Test-Path -LiteralPath $_) } | ForEach-Object { (Resolve-Path -LiteralPath $_).ProviderPath.TrimEnd('\') })
    if ($paths.Count -eq 0) { return }
    $S.Paths = $paths
    $S.Name = if ($paths.Count -eq 1) { Split-Path $paths[0] -Leaf } else { "$($paths.Count) éléments" }
    Set-View 'prep' 'Préparation…' "Analyse de « $($S.Name) » (fichiers cachés compris)…"
    $lblStat.Text = ''; $lblFile.Text = ''
    Start-Bg 'scan' $ScanPaths @(, [string[]]$paths)
    if (-not $S.Croc -and -not $S.Bg.ContainsKey('croc')) { Start-Bg 'croc' $InstallCroc @($AppDir) }
    $S.Scan = $null
}
function Start-SendCroc {
    $list = @($S.Paths)
    if ($S.AttrFile) { $list += $S.AttrFile }
    $q = ($list | ForEach-Object { '"' + $_ + '"' }) -join ' '
    Start-Croc ("--no-compress send --code " + $S.Code + " " + $q) $null
}
function Send-Ready {
    $sc = $S.Scan
    $S.Size = [int64]$sc.Size
    $S.AttrFile = $null
    if ($sc.Attr -and $sc.Attr.Count) {
        $dir = Join-Path $AppDir ('envoi_' + [guid]::NewGuid().ToString('N').Substring(0, 8))
        New-Item -ItemType Directory -Force -Path $dir | Out-Null
        $S.AttrFile = Join-Path $dir 'attributs.tsattr'
        [IO.File]::WriteAllLines($S.AttrFile, [string[]]$sc.Attr, (New-Object Text.UTF8Encoding $false))
    }
    $S.Code = New-Code
    $link = $Site + '#' + $S.Code
    $msg = "Je t'envoie « $($S.Name) » ($(Format-Size $S.Size)), chiffré de bout en bout.`r`nClique sur ce lien pour le recevoir : $link"
    $box.Text = $msg
    $ok = Copy-Text $msg
    $sub = if ($ok) { "✓ Message copié. Colle-le à ton client (Ctrl + V) par mail, SMS ou WhatsApp." } else { "Copie ce message et envoie-le à ton client." }
    if ($sc.Errors -gt 0) { $sub += "`nAttention : $($sc.Errors) élément(s) illisible(s) (accès refusé)." }
    Set-View 'wait' 'Envoie ce message à ton client' $sub
    $lblStat.Text = 'En attente de ton client…'
    $lblFile.Text = "$($sc.Files) fichier(s) · $(Format-Size $S.Size) · le transfert démarre dès qu'il ouvre le lien"
    Set-Buttons 'Copier le message' 'Annuler'
    $S.Retries = 0
    Start-SendCroc
}
function Send-Finished {
    if ($S.AttrFile) { try { Remove-Item -LiteralPath (Split-Path $S.AttrFile -Parent) -Recurse -Force } catch {} ; $S.AttrFile = $null }
    Set-Fill 100
    Set-View 'done' '✓ Transfert terminé' "Ton client a reçu « $($S.Name) » en entier ($(Format-Size $S.Size))."
    Set-Buttons 'Envoyer autre chose' 'Fermer'
}
function Reset-Send {
    Stop-Croc
    if ($S.AttrFile) { try { Remove-Item -LiteralPath (Split-Path $S.AttrFile -Parent) -Recurse -Force } catch {} ; $S.AttrFile = $null }
    Set-View 'idle' 'Envoyer un dossier' 'Glisse ton dossier ci-dessous : le message à envoyer à ton client est copié automatiquement.'
    Set-Buttons '' ''
}
function Install-Shortcut {
    try {
        $cmd = Join-Path $AppDir 'Envoyer.cmd'
        $content = "@echo off`r`nset `"TS_ARGS=%*`"`r`nstart `"`" powershell -NoProfile -WindowStyle Hidden -ExecutionPolicy Bypass -Command `"[Net.ServicePointManager]::SecurityProtocol='Tls12'; iex (irm ${Site}e.txt)`"`r`n"
        [IO.File]::WriteAllText($cmd, $content, (New-Object Text.ASCIIEncoding))
        $desk = [Environment]::GetFolderPath('Desktop')
        $lnkPath = Join-Path $desk 'Transfert sécurisé.lnk'
        if (-not (Test-Path -LiteralPath $lnkPath)) {
            $ws = New-Object -ComObject WScript.Shell
            $lnk = $ws.CreateShortcut($lnkPath)
            $lnk.TargetPath = $cmd
            $lnk.WorkingDirectory = $AppDir
            $lnk.WindowStyle = 7
            $lnk.IconLocation = "$env:SystemRoot\System32\shell32.dll,47"
            $lnk.Description = 'Envoyer un dossier chiffré à un client'
            $lnk.Save()
        }
    } catch {}
}

# ---------------------------------------------------------------- RECEPTION
function Get-CodeFrom([string]$t) {
    if (-not $t) { return '' }
    $m = [regex]::Match($t, '\d{4}(-[a-z0-9]{5}){3}')
    if ($m.Success) { return $m.Value }
    $t = ($t -replace '^(?i)\s*code d.acc[eè]s\s*:\s*', '').Trim()
    return $t
}
function Start-Recv {
    if (-not $S.Code) {
        Set-View 'code' 'Recevoir un dossier' "Colle le lien ou le code d'accès qu'on t'a envoyé."
        Set-Buttons 'Recevoir' ''
        $txtCode.Focus() | Out-Null
        return
    }
    $dest = Join-Path $env:USERPROFILE 'Downloads'
    try { $p = (New-Object -ComObject Shell.Application).Namespace('shell:Downloads').Self.Path; if ($p) { $dest = $p } } catch {}
    if (-not (Test-Path -LiteralPath $dest)) { New-Item -ItemType Directory -Force -Path $dest | Out-Null }
    $S.Dest = $dest
    $S.Total = 0.0; $S.Overall = 0.0
    if (-not $S.Croc) {
        Set-View 'prep' 'Préparation…' 'Première utilisation : installation du module sécurisé (quelques secondes).'
        Set-Buttons '' 'Annuler'
        Start-Bg 'croc' $InstallCroc @($AppDir)
        return
    }
    Start-RecvCroc
}
function Start-RecvCroc {
    Set-View 'connect' 'Connexion à l''expéditeur…' "Le dossier sera enregistré dans : $($S.Dest)"
    Set-Buttons '' 'Annuler'
    $lblStat.Text = 'Connexion sécurisée en cours…'
    $S.StartTime = Get-Date
    Start-Croc ('--yes --overwrite --out "' + $S.Dest + '"') @{ CROC_SECRET = $S.Code }
}
function Recv-Finished {
    try {
        Get-ChildItem -LiteralPath $S.Dest -Filter '*.tsattr' -File -Force | Where-Object { $_.LastWriteTime -ge $S.StartTime.AddMinutes(-1) } | ForEach-Object {
            foreach ($line in [IO.File]::ReadAllLines($_.FullName, [Text.Encoding]::UTF8)) {
                $parts = $line -split "`t", 2
                if ($parts.Count -ne 2 -or $parts[1] -match '(^|\\)\.\.(\\|$)' -or [IO.Path]::IsPathRooted($parts[1])) { continue }
                $p = Join-Path $S.Dest $parts[1]
                if (-not (Test-Path -LiteralPath $p)) { continue }
                try {
                    $item = Get-Item -LiteralPath $p -Force
                    $a = $item.Attributes
                    if ($parts[0] -match 'H') { $a = $a -bor [IO.FileAttributes]::Hidden }
                    if ($parts[0] -match 'S') { $a = $a -bor [IO.FileAttributes]::System }
                    if ($parts[0] -match 'R') { $a = $a -bor [IO.FileAttributes]::ReadOnly }
                    $item.Attributes = $a
                } catch {}
            }
            Remove-Item -LiteralPath $_.FullName -Force
        }
    } catch {}
    Set-Fill 100
    $size = if ($S.Total -gt 0) { " ($(Format-Size $S.Total))" } else { '' }
    Set-View 'done' '✓ Dossier reçu' "Tout est arrivé$size, enregistré dans :`n$($S.Dest)"
    Set-Buttons 'Ouvrir le dossier' 'Fermer'
    try { Start-Process explorer.exe $S.Dest } catch {}
}

# ---------------------------------------------------------------- boucle principale
$timer = New-Object Windows.Forms.Timer
$timer.Interval = 250
$timer.Add_Tick({
    try {
        # animation « en attente »
        if ($S.Indeterminate -and $track.Visible) {
            $S.Anim = ($S.Anim + 8) % ($track.Width + 130)
            $fill.Width = 130; $fill.Left = $S.Anim - 130
        }
        # tâches de fond
        if ($S.Bg.ContainsKey('croc')) {
            $r = Receive-Bg 'croc'
            if ($r) {
                if ($r.Ok) { $S.Croc = [string]$r.Value } else { Show-Error ("Impossible d'installer le module de transfert. Vérifie la connexion internet.`n" + $r.Error) ; return }
                if ($Mode -eq 'recv' -and $S.View -eq 'prep') { Start-RecvCroc }
            }
        }
        if ($S.Bg.ContainsKey('scan')) {
            $r = Receive-Bg 'scan'
            if ($r) { if ($r.Ok) { $S.Scan = $r.Value } else { Show-Error ("Impossible de lire le dossier.`n" + $r.Error); return } }
        }
        if ($Mode -eq 'send' -and $S.View -eq 'prep' -and $S.Scan -and $S.Croc) { Send-Ready }

        # nouvelle tentative programmée
        if ($S.RetryAt -and (Get-Date) -ge $S.RetryAt) {
            $S.RetryAt = $null
            if ($Mode -eq 'send') { Start-SendCroc } else { Start-RecvCroc }
        }

        # sortie du moteur
        if ($S.Running -and $S.Proc) {
            Read-Croc
            if ($S.Proc -and $S.Proc.HasExited -and -not ($S.Readers | Where-Object { -not $_.Done })) {
                $code = $S.Proc.ExitCode
                $S.Running = $false; $S.Proc = $null; Set-Awake $false
                if ($S.View -eq 'error') { return }
                if ($code -eq 0) {
                    if ($Mode -eq 'send') { Send-Finished } else { Recv-Finished }
                } else {
                    $S.Retries++
                    $max = if ($Mode -eq 'send') { 200 } else { 120 }
                    if ($S.Retries -gt $max) {
                        $last = ($S.Lines | Select-Object -Last 3) -join "`n"
                        if ($Mode -eq 'send') { Show-Error "La connexion n'a pas pu être rétablie.`n$last" }
                        else { Show-Error "Impossible de joindre l'expéditeur. Vérifie que sa fenêtre est toujours ouverte, puis clique sur Réessayer.`n$last" }
                        return
                    }
                    $S.RetryAt = (Get-Date).AddSeconds(4)
                    $title = if ($S.GotProgress) { 'Connexion perdue, reprise automatique…' } elseif ($Mode -eq 'send') { 'Envoie ce message à ton client' } else { 'Connexion à l''expéditeur…' }
                    if ($S.GotProgress -or $Mode -eq 'recv') {
                        $keepSub = $lblSub.Text
                        Set-View 'retry' $title ($(if ($S.GotProgress) { 'Le transfert reprendra exactement là où il s''est arrêté.' } else { $keepSub }))
                        Set-Buttons '' 'Annuler'
                        $lblStat.Text = "Nouvel essai dans quelques secondes (essai $($S.Retries))…"
                    }
                }
            }
        }
    } catch {
        $timer.Stop()
        [Windows.Forms.MessageBox]::Show("Erreur inattendue : $($_.Exception.Message)", 'Transfert sécurisé') | Out-Null
        $timer.Start()
    }
})
